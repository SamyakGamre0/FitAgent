    override fun onCreate(savedInstanceState: Bundle?) {

        setContentView(R.layout.activity_history)

        val recycler = findViewById<RecyclerView>(R.id.recycler_history)

        recycler.layoutManager = LinearLayoutManager(this)

            } else {
                tvEmpty.visibility = View.GONE
                recycler.visibility = View.VISIBLE
                recycler.adapter = HistoryAdapter(sessions)
            }
        }
    }
}

package com.FitAgent

import android.os.Bundle
import android.os.Handler
import android.os.Looper
import android.widget.TextView
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import androidx.lifecycle.lifecycleScope
import androidx.recyclerview.widget.LinearLayoutManager
import androidx.recyclerview.widget.RecyclerView
import com.FitAgent.adapters.LoggedExercise
import com.FitAgent.adapters.LoggerAdapter
import com.FitAgent.database.AppDatabase
import com.FitAgent.models.ExerciseType
import com.FitAgent.models.WorkoutLog
import com.FitAgent.models.WorkoutSession
import com.FitAgent.utils.WorkoutGenerator
import com.google.android.material.button.MaterialButton
import kotlinx.coroutines.launch
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

class LoggerActivity : AppCompatActivity() {

    private lateinit var tvTimer: TextView
    private var seconds = 0
    private val handler = Handler(Looper.getMainLooper())
    private lateinit var timerRunnable: Runnable
    private lateinit var loggedList: MutableList<LoggedExercise>

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_logger)

        tvTimer = findViewById(R.id.tv_timer)

        val goal = intent.getStringExtra("GOAL") ?: "General"
        val equipment = intent.getStringExtra("EQUIPMENT") ?: "None"

        val exercises = WorkoutGenerator.generate(goal, equipment)
        loggedList = exercises.map { LoggedExercise(it, sets = it.sets) }.toMutableList()

        val recycler = findViewById<RecyclerView>(R.id.recycler_logger)
        recycler.layoutManager = LinearLayoutManager(this)
        recycler.adapter = LoggerAdapter(loggedList)

        startTimer()

        val btnSave = findViewById<MaterialButton>(R.id.btn_save_workout)
        btnSave.setOnClickListener {
            saveWorkoutToDatabase(goal, equipment)
        }
    }

    private fun saveWorkoutToDatabase(goal: String, equipment: String) {
        val validLogged = loggedList.filter { logged ->
            when (logged.exercise.type) {
                ExerciseType.WEIGHTED -> logged.weight > 0 && logged.reps > 0
                ExerciseType.BODYWEIGHT -> logged.reps > 0
                ExerciseType.TIMED -> logged.seconds > 0
            }
        }

        if (validLogged.isEmpty()) {
            Toast.makeText(this,
                "Please log at least one exercise",
                Toast.LENGTH_SHORT).show()
            return
        }

        val totalVolume = validLogged.sumOf { logged ->
            when (logged.exercise.type) {
                ExerciseType.WEIGHTED -> logged.weight * logged.reps * logged.sets
                ExerciseType.BODYWEIGHT -> (logged.reps * logged.sets).toDouble()
                ExerciseType.TIMED -> (logged.seconds * logged.sets).toDouble()
            }
        }

        val mins = seconds / 60
        val secs = seconds % 60
        val duration = String.format("%02d:%02d", mins, secs)
        val date = SimpleDateFormat("dd MMM yyyy, HH:mm", Locale.getDefault())
            .format(Date())

        val session = WorkoutSession(
            date = date,
            goal = goal,
            equipment = equipment,
            duration = duration,
            totalVolume = totalVolume,
            exerciseCount = validLogged.size
        )

        val db = AppDatabase.getDatabase(this)

        lifecycleScope.launch {
            val sessionId = db.workoutDao().insertSession(session)

            validLogged.forEach { logged ->
                val (weight, reps, unit) = when (logged.exercise.type) {
                    ExerciseType.WEIGHTED -> Triple(logged.weight, logged.reps, logged.unit)
                    ExerciseType.BODYWEIGHT -> Triple(0.0, logged.reps, "reps")
                    ExerciseType.TIMED -> Triple(0.0, logged.seconds, "sec")
                }

                val volume = when (logged.exercise.type) {
                    ExerciseType.WEIGHTED -> logged.weight * logged.reps * logged.sets
                    ExerciseType.BODYWEIGHT -> (logged.reps * logged.sets).toDouble()
                    ExerciseType.TIMED -> (logged.seconds * logged.sets).toDouble()
                }

                val log = WorkoutLog(
                    sessionId = sessionId,
                    exerciseName = logged.exercise.name,
                    muscleGroup = logged.exercise.muscleGroup,
                    sets = logged.sets,
                    weight = weight,
                    reps = reps,
                    unit = unit,
                    volume = volume
                )
                db.workoutDao().insertLog(log)
            }

            Toast.makeText(this@LoggerActivity,
                "✅ Workout Saved!",
                Toast.LENGTH_LONG).show()

            handler.removeCallbacks(timerRunnable)
            finish()
        }
    }

    private fun startTimer() {
        timerRunnable = object : Runnable {
            override fun run() {
                seconds++
                val mins = seconds / 60
                val secs = seconds % 60
                tvTimer.text = String.format("⏱ %02d:%02d", mins, secs)
                handler.postDelayed(this, 1000)
            }
        }
        handler.post(timerRunnable)
    }

    override fun onDestroy() {
        super.onDestroy()
        handler.removeCallbacks(timerRunnable)
    }
}

package com.FitAgent

import android.content.Intent
import android.os.Bundle
import android.widget.ArrayAdapter
import android.widget.Spinner
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import com.google.android.material.button.MaterialButton

class MainActivity : AppCompatActivity() {

    private lateinit var spinnerGoal: Spinner
    private lateinit var spinnerEquipment: Spinner
    private lateinit var spinnerDays: Spinner
    private lateinit var btnGenerate: MaterialButton

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        spinnerGoal = findViewById(R.id.spinner_goal)
        spinnerEquipment = findViewById(R.id.spinner_equipment)
        spinnerDays = findViewById(R.id.spinner_days)
        btnGenerate = findViewById(R.id.btn_generate)

        setupSpinners()

        btnGenerate.setOnClickListener {
            handleGenerateClick()
        }

        val btnHistory = findViewById<MaterialButton>(R.id.btn_view_history)
        btnHistory?.setOnClickListener {
            startActivity(Intent(this, HistoryActivity::class.java))
        }
    }

    private fun setupSpinners() {
        val goals = arrayOf(
            "Choose your goal...",
            "Muscle Gain 💪",
            "Weight Loss 🔥",
            "General Fitness ⚡",
            "Build Endurance 🏃"
        )
        val goalsAdapter = ArrayAdapter(this,
            android.R.layout.simple_spinner_item, goals)
        goalsAdapter.setDropDownViewResource(
            android.R.layout.simple_spinner_dropdown_item)
        spinnerGoal.adapter = goalsAdapter

        val equipment = arrayOf(
            "Choose equipment...",
            "No Equipment 🏠",
            "Dumbbells Only 🏋️",
            "Full Gym 🏟️",
            "Resistance Bands 🎽"
        )
        val equipAdapter = ArrayAdapter(this,
            android.R.layout.simple_spinner_item, equipment)
        equipAdapter.setDropDownViewResource(
            android.R.layout.simple_spinner_dropdown_item)
        spinnerEquipment.adapter = equipAdapter

        val days = arrayOf(
            "Select days...",
            "3 days",
            "4 days",
            "5 days",
            "6 days"
        )
        val daysAdapter = ArrayAdapter(this,
            android.R.layout.simple_spinner_item, days)
        daysAdapter.setDropDownViewResource(
            android.R.layout.simple_spinner_dropdown_item)
        spinnerDays.adapter = daysAdapter
    }

    private fun handleGenerateClick() {
        val selectedGoal = spinnerGoal.selectedItem.toString()
        val selectedEquipment = spinnerEquipment.selectedItem.toString()
        val selectedDays = spinnerDays.selectedItem.toString()

        if (selectedGoal.startsWith("Choose") ||
            selectedEquipment.startsWith("Choose") ||
            selectedDays.startsWith("Select")) {
            Toast.makeText(this,
                "Please select all options",
                Toast.LENGTH_SHORT).show()
            return
        }

        val intent = Intent(this, WorkoutPlanActivity::class.java)
        intent.putExtra("GOAL", selectedGoal)
        intent.putExtra("EQUIPMENT", selectedEquipment)
        startActivity(intent)
    }
}

import android.text.Editable
import android.text.TextWatcher
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import android.widget.LinearLayout
import android.widget.TextView
import androidx.recyclerview.widget.RecyclerView

import com.google.android.material.button.MaterialButton
import com.google.android.material.textfield.TextInputEditText
import com.google.android.material.textfield.TextInputLayout

data class LoggedExercise(
    val exercise: Exercise,
    var weight: Double = 0.0,
    var reps: Int = 0,
    var seconds: Int = 0,
    var sets: Int = 3,
    var unit: String = "kg"
)

class LoggerAdapter(
    val loggedList: MutableList<LoggedExercise>
) : RecyclerView.Adapter<LoggerAdapter.LoggerViewHolder>() {

    class LoggerViewHolder(view: View) : RecyclerView.ViewHolder(view) {
        val tvName: TextView = view.findViewById(R.id.tv_log_exercise_name)
        val tvMuscle: TextView = view.findViewById(R.id.tv_log_muscle)
        val btnMinus: MaterialButton = view.findViewById(R.id.btn_minus_set)
        val btnPlus: MaterialButton = view.findViewById(R.id.btn_plus_set)
        val tvSets: TextView = view.findViewById(R.id.tv_sets_count)

        val layoutWeighted: LinearLayout = view.findViewById(R.id.layout_weighted)
        val layoutBodyweight: TextInputLayout = view.findViewById(R.id.layout_bodyweight)
        val layoutTimed: TextInputLayout = view.findViewById(R.id.layout_timed)

        val etWeight: TextInputEditText = view.findViewById(R.id.et_weight)
        val btnUnit: MaterialButton = view.findViewById(R.id.btn_unit_toggle)
        val etRepsWeighted: TextInputEditText = view.findViewById(R.id.et_reps_weighted)
        val etRepsBodyweight: TextInputEditText = view.findViewById(R.id.et_reps_bodyweight)
        val etSeconds: TextInputEditText = view.findViewById(R.id.et_seconds)
    }

    override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): LoggerViewHolder {
        val view = LayoutInflater.from(parent.context)
            .inflate(R.layout.item_logger, parent, false)
        return LoggerViewHolder(view)
    }

    override fun onBindViewHolder(holder: LoggerViewHolder, position: Int) {
        val logged = loggedList[position]
        holder.tvName.text = logged.exercise.name
        holder.tvMuscle.text = "Target: ${logged.exercise.muscleGroup}"
        holder.tvSets.text = logged.sets.toString()
        holder.btnUnit.text = logged.unit

        holder.layoutWeighted.visibility = View.GONE
        holder.layoutBodyweight.visibility = View.GONE
        holder.layoutTimed.visibility = View.GONE

        when (logged.exercise.type) {
            ExerciseType.WEIGHTED -> holder.layoutWeighted.visibility = View.VISIBLE
            ExerciseType.BODYWEIGHT -> holder.layoutBodyweight.visibility = View.VISIBLE
            ExerciseType.TIMED -> holder.layoutTimed.visibility = View.VISIBLE
        }

        holder.btnMinus.setOnClickListener {
            if (logged.sets > 1) {
                logged.sets--
                holder.tvSets.text = logged.sets.toString()
            }
        }
        holder.btnPlus.setOnClickListener {
            if (logged.sets < 10) {
                logged.sets++
                holder.tvSets.text = logged.sets.toString()
            }
        }
        holder.btnUnit.setOnClickListener {
            logged.unit = if (logged.unit == "kg") "lbs" else "kg"
            holder.btnUnit.text = logged.unit
        }

        holder.etWeight.addTextChangedListener(simpleWatcher {
            logged.weight = it.toDoubleOrNull() ?: 0.0
        })
        holder.etRepsWeighted.addTextChangedListener(simpleWatcher {
            logged.reps = it.toIntOrNull() ?: 0
        })
        holder.etRepsBodyweight.addTextChangedListener(simpleWatcher {
            logged.reps = it.toIntOrNull() ?: 0
        })
        holder.etSeconds.addTextChangedListener(simpleWatcher {
            logged.seconds = it.toIntOrNull() ?: 0
        })
    }

    private fun simpleWatcher(action: (String) -> Unit) = object : TextWatcher {
        override fun beforeTextChanged(s: CharSequence?, start: Int, count: Int, after: Int) {}
        override fun onTextChanged(s: CharSequence?, start: Int, before: Int, count: Int) {}
        override fun afterTextChanged(s: Editable?) { action(s.toString()) }
    }

    override fun getItemCount() = loggedList.size
}

package com.FitAgent.database

import android.content.Context
import androidx.room.Database
import androidx.room.Room
import androidx.room.RoomDatabase
import com.FitAgent.models.WorkoutLog
import com.FitAgent.models.WorkoutSession

@Database(
    entities = [WorkoutSession::class, WorkoutLog::class],
    version = 2,
    exportSchema = false
)
abstract class AppDatabase : RoomDatabase() {

    abstract fun workoutDao(): WorkoutDao

    companion object {
        @Volatile
        private var INSTANCE: AppDatabase? = null

        fun getDatabase(context: Context): AppDatabase {
            return INSTANCE ?: synchronized(this) {
                val instance = Room.databaseBuilder(
                    context.applicationContext,
                    AppDatabase::class.java,
                    "fitagent_database"
                ).fallbackToDestructiveMigration().build()
                INSTANCE = instance
                instance
            }
        }
    }
}

import androidx.room.Query

@Dao
interface WorkoutDao {

    @Insert
    suspend fun insertSession(session: WorkoutSession): Long

    @Insert
    suspend fun insertLog(log: WorkoutLog)

    @Query("SELECT * FROM workout_sessions ORDER BY id DESC")
    suspend fun getAllSessions(): List<WorkoutSession>

    @Query("SELECT * FROM workout_logs WHERE sessionId = :sessionId")
    suspend fun getLogsForSession(sessionId: Long): List<WorkoutLog>

    @Query("SELECT COUNT(*) FROM workout_sessions")
    suspend fun getTotalWorkouts(): Int

    @Query("SELECT SUM(totalVolume) FROM workout_sessions")
    suspend fun getTotalVolumeLifted(): Double?
}

package com.FitAgent.utils

import com.FitAgent.models.Exercise

object WorkoutGenerator {

        val equipKey = when {
            equipment.contains("No Equipment", true) -> "none"
            equipment.contains("Dumbbell", true) -> "dumbbell"
            equipment.contains("Full Gym", true) -> "gym"
            equipment.contains("Resistance", true) -> "bands"
            else -> "none"
        }

        return when ("$goalKey-$equipKey") {

            // MUSCLE GAIN
            "muscle-none" -> listOf(
                Exercise("Push-ups", 4, "12-15", "Chest", "Keep core tight", ExerciseType.BODYWEIGHT),
                Exercise("Diamond Push-ups", 3, "10", "Triceps", "Form diamond with hands", ExerciseType.BODYWEIGHT),
                Exercise("Bodyweight Squats", 4, "20", "Legs", "Go deep", ExerciseType.BODYWEIGHT),
                Exercise("Pike Push-ups", 3, "10", "Shoulders", "Elevate hips high", ExerciseType.BODYWEIGHT),
                Exercise("Plank", 3, "60", "Core", "Straight body line", ExerciseType.TIMED)

            "muscle-dumbbell" -> listOf(
